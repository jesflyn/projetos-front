Envie um arquivo .csv para visualizar o head, tipos, valores ausentes e estatísticas básicas — tudo rodando com pandas via PyScript.
Dica: separador padrão é vírgula. Se seu arquivo usa ponto-e-vírgula, altere o campo "Separador" abaixo antes de enviar.

from js import document, URL, Blob from pyodide.ffi import create_proxy import pandas as pd import io _df = None # armazenará o último DataFrame carregado # Utilitários para escrever HTML de forma segura from pyscript import display def clear_targets(): for tid in ["preview", "dtypes", "missing", "stats"]: document.getElementById(tid).innerHTML = "" document.getElementById("shape").innerText = "—" async def on_file_change(event): # Habilita o botão de analisar quando um arquivo for selecionado btn = document.getElementById("btn-analisar") btn.disabled = False async def analyze_click(event): global _df file_input = document.getElementById("csvfile") if not file_input.files.length: return file = file_input.files.item(0) sep = document.getElementById("sep").value or "," encoding = document.getElementById("encoding").value or "utf-8" # Lê conteúdo do arquivo (como texto) e carrega no pandas text = await file.text() try: _df = pd.read_csv(io.StringIO(text), sep=sep, encoding=encoding) except Exception: # Alguns navegadores podem fornecer texto já decodificado; tente sem 'encoding' _df = pd.read_csv(io.StringIO(text), sep=sep) clear_targets() # Exibe dimensões document.getElementById("shape").innerText = f"{_df.shape[0]} linhas × {_df.shape[1]} colunas" # Prévia display(_df.head(), target="preview") # dtypes dtypes = _df.dtypes.to_frame("dtype").reset_index(names=["coluna"]) display(dtypes, target="dtypes") # missing por coluna missing = _df.isna().sum().to_frame("faltantes").reset_index(names=["coluna"]) display(missing, target="missing") # estatísticas stats = _df.describe(include="all", datetime_is_numeric=True) display(stats, target="stats") # Habilita baixar head document.getElementById("btn-baixar").disabled = False async def download_head(event): if _df is None: return head_csv = _df.head().to_csv(index=False) blob = Blob.new([head_csv], {"type": "text/csv"}) url = URL.createObjectURL(blob) link = document.createElement("a") link.href = url link.download = "head.csv" document.body.appendChild(link) link.click() document.body.removeChild(link) URL.revokeObjectURL(url) # liga eventos input_el = document.getElementById("csvfile") input_el.addEventListener("change", create_proxy(on_file_change)) document.getElementById("btn-analisar").addEventListener("click", create_proxy(analyze_click)) document.getElementById("btn-baixar").addEventListener("click", create_proxy(download_head))
